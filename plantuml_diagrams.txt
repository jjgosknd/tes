  Глава 3: Рис.8-9 (экраны приложения)

==============================================================
Рис. 1. Функциональная схема работы мобильного приложения EduFlow
==============================================================

@startuml
skinparam backgroundColor white
skinparam defaultFontName "Times New Roman"
skinparam defaultFontSize 11

start
:Запуск приложения;
if (Токен сохранён?) then (да)
  :Автоматический вход\n(refresh token);
else (нет)
  :Экран авторизации;
  :Ввод email и пароля;
  :POST /api/v1/auth/login;
  :Сохранение JWT-токенов\nв EncryptedSharedPreferences;
endif

:Загрузка каталога курсов\nGET /api/v1/categories\nGET /api/v1/courses;

:Отображение списка курсов\nс фильтрацией по категориям;

:Выбор курса пользователем;
:Загрузка карточки курса\nGET /api/v1/courses/{id};

if (Пользователь записан?) then (нет)
  :Запись на курс\nPOST /api/v1/courses/{id}/enroll;
else (да)
endif

:Последовательное прохождение уроков;

repeat
  :Загрузка урока\nGET /api/v1/lessons/{id};
  :Просмотр видео и текста;
  :Прохождение теста\nPOST /api/v1/lessons/{id}/quiz;
  :Отметка урока пройденным\nPOST /api/v1/lessons/{id}/complete;
repeat while (Ещё есть уроки?) is (да)
->нет;

:Генерация сертификата\nна стороне сервера;
:Отображение в профиле\nGET /api/v1/me/certificates;
:Загрузка PDF\nGET /api/v1/me/certificates/{id}/download;

stop
@enduml

Рис. 6. Диаграмма классов мобильного клиента
==============================================================

@startuml
skinparam backgroundColor white
skinparam defaultFontName "Times New Roman"
skinparam defaultFontSize 10
skinparam classBackgroundColor #DEEBF7
skinparam classBorderColor #2E75B6

package "UI Layer (Jetpack Compose)" {
  class LoginScreen
  class CatalogScreen
  class CourseDetailsScreen
  class LessonScreen
  class ProfileScreen
}

package "ViewModel Layer" #FFF2CC {
  class LoginViewModel {
    - authRepo: AuthRepository
    + submit(email, password)
    + state: StateFlow<LoginUiState>
  }
  class CatalogViewModel {
    - catalogRepo: CatalogRepository
    + load()
    + filter(categoryId)
    + state: StateFlow<CatalogUiState>
  }
  class LessonViewModel {
    - learningRepo: LearningRepository
    + load(lessonId)
    + complete()
    + submitQuiz(answers)
    + state: StateFlow<LessonUiState>
  }
  class ProfileViewModel {
    - certRepo: CertificateRepository
    + loadProfile()
    + downloadPdf(certId)
  }
}

package "Data Layer (Repository)" #E2EFDA {
  interface AuthRepository {
    + login(email, pwd): ApiResult
    + register(email, pwd, name): ApiResult
    + logout()
    + refresh(): ApiResult
  }
  interface CatalogRepository {
    + categories(): List<Category>
    + courses(filter): List<CourseSummary>
    + courseById(id): CourseDetails
  }
  interface LearningRepository {
    + enroll(courseId)
    + lesson(id): LessonContent
    + complete(lessonId)
    + submitQuiz(lessonId, answers): QuizResult
    + myCourses(): List<MyCourse>
  }
  interface CertificateRepository {
    + listMine(): List<Certificate>
    + downloadPdf(id): ByteArray
  }
}

package "Network Layer" #FCE4D6 {
  interface EduFlowApi <<Retrofit>>
  interface AuthApi <<Retrofit>>
  class EncryptedTokenStore
  class AuthInterceptor
  class TokenAuthenticator
}

LoginScreen --> LoginViewModel
CatalogScreen --> CatalogViewModel
LessonScreen --> LessonViewModel
ProfileScreen --> ProfileViewModel

LoginViewModel --> AuthRepository
CatalogViewModel --> CatalogRepository
LessonViewModel --> LearningRepository
ProfileViewModel --> CertificateRepository

AuthRepository --> AuthApi
CatalogRepository --> EduFlowApi
LearningRepository --> EduFlowApi
CertificateRepository --> EduFlowApi

AuthInterceptor --> EncryptedTokenStore
TokenAuthenticator --> AuthApi
@enduml

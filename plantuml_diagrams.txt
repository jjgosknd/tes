Рис. 1. Функциональная схема работы мобильного приложения EduFlow
(Блок-схема общего алгоритма работы — как у друга рис.1)
==============================================================

@startuml
skinparam backgroundColor white
skinparam ActivityBackgroundColor #DEEBF7
skinparam ActivityBorderColor #2E75B6
skinparam DiamondBackgroundColor #FFF2CC
skinparam DiamondBorderColor #BF9000

|Мобильный клиент|
start
:Запуск приложения EduFlow;
if (Сохранён refresh-токен?) then (да)
  :Отправка POST /api/v1/auth/refresh;
  if (Токен валиден?) then (да)
    :Получение нового access-токена;
  else (нет)
    :Переход на экран авторизации;
    :Ввод email и пароля;
    :POST /api/v1/auth/login;
    :Сохранение JWT-токенов в\nEncryptedSharedPreferences;
  endif
else (нет)
  :Экран авторизации/регистрации;
  :POST /api/v1/auth/login или /register;
  :Сохранение JWT-токенов;
endif

:Загрузка каталога курсов\nGET /api/v1/categories\nGET /api/v1/courses;
:Отображение списка курсов\nс фильтрацией по категориям;

:Пользователь выбирает курс;
:Загрузка карточки курса\nGET /api/v1/courses/{id};

if (Пользователь записан на курс?) then (нет)
  :Запись на курс\nPOST /api/v1/courses/{id}/enroll;
else (да)
endif

|Цикл прохождения уроков|
repeat
  :Загрузка содержимого урока\nGET /api/v1/lessons/{id};
  :Отображение видеоплеера\n(Media3 ExoPlayer) и текста;
  :Отображение проверочного теста;
  :Пользователь отвечает на вопросы;
  :Отправка ответов\nPOST /api/v1/lessons/{id}/quiz;
  if (Тест пройден >= 70%?) then (да)
    :Отметка урока пройденным\nPOST /api/v1/lessons/{id}/complete;
  else (нет)
    :Сообщение «Попробуйте ещё раз»;
  endif
repeat while (Есть непройденные уроки?) is (да)
->нет;

|Мобильный клиент|
:Сервер генерирует PDF-сертификат;
:Отображение сертификата в профиле;
:Пользователь скачивает PDF\nGET /api/v1/me/certificates/{id}/download;

stop
@enduml

@startuml
skinparam backgroundColor white
skinparam classBackgroundColor #DEEBF7

package "UI Layer (Jetpack Compose Screens)" {
  class LoginScreen << @Composable >>
  class CatalogScreen << @Composable >>
  class CourseDetailsScreen << @Composable >>
  class LessonScreen << @Composable >>
  class MyCoursesScreen << @Composable >>
  class ProfileScreen << @Composable >>
}

package "ViewModel Layer (androidx.lifecycle)" #FFF2CC {
  class LoginViewModel {
    - authRepo: AuthRepository
    - _state: MutableStateFlow<LoginUiState>
    + state: StateFlow<LoginUiState>
    + submit(email: String, password: String)
  }
  class CatalogViewModel {
    - catalogRepo: CatalogRepository
    - _state: MutableStateFlow<CatalogUiState>
    + state: StateFlow<CatalogUiState>
    + load()
    + onCategorySelected(id: Long?)
    + onQueryChanged(q: String)
  }
  class LessonViewModel {
    - learningRepo: LearningRepository
    - _state: MutableStateFlow<LessonUiState>
    + state: StateFlow<LessonUiState>
    + load(lessonId: Long)
    + complete()
    + submitQuiz(answers: Map<Long, Int>)
  }
  class ProfileViewModel {
    - certRepo: CertificateRepository
    - meRepo: MeRepository
    + loadProfile()
    + downloadPdf(certId: Long)
  }
}

package "Data Layer (Repository)" #E2EFDA {
  interface AuthRepository {
    + login(email, pwd): Flow<ApiResult<AuthUser>>
    + register(email, pwd, name): Flow<ApiResult<AuthUser>>
    + logout()
    + currentUser: Flow<AuthUser?>
  }
  interface CatalogRepository {
    + categories(): ApiResult<List<Category>>
    + courses(categoryId, level, query): ApiResult<List<CourseSummary>>
    + courseById(id): ApiResult<CourseDetails>
  }
  interface LearningRepository {
    + enroll(courseId): ApiResult<Unit>
    + lesson(id): ApiResult<LessonContent>
    + completeLesson(id): ApiResult<Unit>
    + submitQuiz(id, answers): ApiResult<QuizResult>
    + myCourses(): ApiResult<List<MyCourse>>
  }
  interface CertificateRepository {
    + listMine(): ApiResult<List<Certificate>>
    + downloadPdf(id): ApiResult<ByteArray>
  }
}

package "Network Layer (Retrofit + OkHttp)" #FCE4D6 {
  interface EduFlowApi << @GET, @POST >> {
    + categories(): CategoryListResponse
    + courses(category, level, q): CourseListResponse
    + courseById(id): CourseDetailsDto
    + enroll(id): EnrollmentDto
    + lesson(id): LessonContentDto
    + completeLesson(id): LessonProgressDto
    + submitQuiz(id, body): QuizResultDto
    + myCourses(): MyCoursesResponse
    + myCertificates(): CertificateListResponse
  }
  interface AuthApi << @POST >> {
    + login(body): AuthResponse
    + register(body): AuthResponse
    + refresh(body): TokensDto
    + logout(body)
  }
  class EncryptedTokenStore {
    + save(user, tokens)
    + access(): Flow<String?>
    + refresh(): Flow<String?>
    + clear()
  }
  class AuthInterceptor << Interceptor >> {
    + intercept(chain): Response
  }
  class TokenAuthenticator << Authenticator >> {
    + authenticate(route, response): Request?
  }
}

LoginScreen ..> LoginViewModel
CatalogScreen ..> CatalogViewModel
LessonScreen ..> LessonViewModel
ProfileScreen ..> ProfileViewModel

LoginViewModel --> AuthRepository
CatalogViewModel --> CatalogRepository
LessonViewModel --> LearningRepository
ProfileViewModel --> CertificateRepository

AuthRepository --> AuthApi
CatalogRepository --> EduFlowApi
LearningRepository --> EduFlowApi
CertificateRepository --> EduFlowApi

AuthInterceptor --> EncryptedTokenStore : "читает access-токен"
TokenAuthenticator --> AuthApi : "POST /auth/refresh"
TokenAuthenticator --> EncryptedTokenStore : "обновляет токены"
@enduml

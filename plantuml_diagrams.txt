Рис. 6. ER-диаграмма базы данных

@startuml
skinparam backgroundColor white

entity "users" {
  *id : BIGINT <<PK>>
  --
  email : VARCHAR
  password_hash : VARCHAR
  full_name : VARCHAR
  role : user_role
  created_at : TIMESTAMP
}

entity "categories" {
  *id : BIGINT <<PK>>
  --
  name : VARCHAR
  slug : VARCHAR
  icon_url : VARCHAR
}

entity "courses" {
  *id : BIGINT <<PK>>
  --
  *category_id : BIGINT <<FK>>
  title : VARCHAR
  description : TEXT
  level : prog_level
  author : VARCHAR
  cover_url : VARCHAR
  duration_min : INT
  is_published : BOOL
}

entity "modules" {
  *id : BIGINT <<PK>>
  --
  *course_id : BIGINT <<FK>>
  title : VARCHAR
  position : INT
}

entity "lessons" {
  *id : BIGINT <<PK>>
  --
  *module_id : BIGINT <<FK>>
  title : VARCHAR
  content_md : TEXT
  video_url : VARCHAR
  position : INT
}

entity "quiz_questions" {
  *id : BIGINT <<PK>>
  --
  *lesson_id : BIGINT <<FK>>
  question : TEXT
  options : JSONB
  correct_idx : INT
}

entity "enrollments" {
  *id : BIGINT <<PK>>
  --
  *user_id : BIGINT <<FK>>
  *course_id : BIGINT <<FK>>
  enrolled_at : TIMESTAMP
}

entity "lesson_progress" {
  *id : BIGINT <<PK>>
  --
  *user_id : BIGINT <<FK>>
  *lesson_id : BIGINT <<FK>>
  completed_at : TIMESTAMP
}

entity "certificates" {
  *id : BIGINT <<PK>>
  --
  *user_id : BIGINT <<FK>>
  *course_id : BIGINT <<FK>>
  issued_at : TIMESTAMP
  certificate_no : VARCHAR
  pdf_url : VARCHAR
}

entity "refresh_tokens" {
  *id : BIGINT <<PK>>
  --
  *user_id : BIGINT <<FK>>
  token_hash : VARCHAR
  expires_at : TIMESTAMP
}

categories ||--o{ courses
courses ||--o{ modules
modules ||--o{ lessons
lessons ||--o{ quiz_questions
users ||--o{ enrollments
courses ||--o{ enrollments
users ||--o{ lesson_progress
lessons ||--o{ lesson_progress
users ||--o{ certificates
courses ||--o{ certificates
users ||--o{ refresh_tokens
@enduml

@startuml
skinparam backgroundColor white

package "UI (Compose Screens)" #DEEBF7 {
  class LoginScreen
  class CatalogScreen
  class CourseDetailsScreen
  class LessonScreen
  class ProfileScreen
}

package "ViewModel" #FFF2CC {
  class LoginViewModel {
    -repo: AuthRepository
    +submit(email, pwd)
  }
  class CatalogViewModel {
    -repo: CatalogRepository
    +load()
    +filter(category)
  }
  class CourseDetailsViewModel {
    -repo: CatalogRepository
    +load(id)
    +enroll()
  }
  class LessonViewModel {
    -repo: LearningRepository
    +load(id)
    +complete()
    +submitQuiz()
  }
  class ProfileViewModel {
    -repo: CertificateRepository
    +loadProfile()
    +downloadPdf(id)
  }
}

package "Repository" #E2EFDA {
  interface AuthRepository {
    +login()
    +register()
    +logout()
  }
  interface CatalogRepository {
    +categories()
    +courses(filter)
    +courseById(id)
  }
  interface LearningRepository {
    +enroll(courseId)
    +lesson(id)
    +complete(id)
    +submitQuiz()
  }
  interface CertificateRepository {
    +listMine()
    +downloadPdf(id)
  }
}

package "Network" #FCE4D6 {
  interface EduFlowApi
  interface AuthApi
  class TokenStore
  class AuthInterceptor
  class TokenAuthenticator
}

Рис. 10. Диаграмма последовательности: авторизация
==============================================================

@startuml
skinparam backgroundColor white

actor Учащийся
participant LoginScreen
participant LoginViewModel
participant AuthRepository
participant AuthApi
participant TokenStore
participant Сервер

Учащийся -> LoginScreen : ввод email/password
LoginScreen -> LoginViewModel : submit(email, pwd)
LoginViewModel -> AuthRepository : login(email, pwd)
AuthRepository -> AuthApi : POST /auth/login
AuthApi -> Сервер : HTTP POST
Сервер --> AuthApi : 200 {access, refresh, user}
AuthApi --> AuthRepository : AuthResponse
AuthRepository -> TokenStore : save(tokens)
TokenStore --> AuthRepository : ok
AuthRepository --> LoginViewModel : Success
LoginViewModel --> LoginScreen : переход в Catalog

... позже: access истёк ...

LoginViewModel -> AuthRepository : GET /courses (401)
AuthRepository -> TokenStore : refresh()
TokenStore -> AuthApi : POST /auth/refresh
AuthApi -> Сервер : refresh token
Сервер --> AuthApi : новый access
AuthApi --> TokenStore : save
note right: запрос повторяется автоматически
@enduml

  :Прохождение урока;
